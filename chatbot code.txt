const msgEl = document.getElementById("messages");

function addMessage(role, content) {
  const row = document.createElement("div");
  row.className = "msg-row " + role;
  const avatar = document.createElement("div");
  avatar.className = "avatar " + (role === "bot" ? "bot" : "user");
  avatar.textContent = role === "bot" ? "🤖" : "👤";
  const bubble = document.createElement("div");
  bubble.className = "bubble " + (role === "bot" ? "bot" : "user");
  bubble.innerHTML = content.replace(/\n/g,"<br>").replace(/\*\*(.*?)\*\*/g,"<strong>$1</strong>");
  if (role === "bot") { row.appendChild(avatar); row.appendChild(bubble); }
  else { row.appendChild(bubble); row.appendChild(avatar); }
  msgEl.appendChild(row);
  msgEl.scrollTop = msgEl.scrollHeight;
}

function addTyping() {
  const row = document.createElement("div");
  row.className = "msg-row bot"; row.id = "typing";
  const avatar = document.createElement("div");
  avatar.className = "avatar bot"; avatar.textContent = "🤖";
  const bubble = document.createElement("div");
  bubble.className = "bubble bot";
  bubble.innerHTML = '<div class="typing"><span></span><span></span><span></span></div>';
  row.appendChild(avatar); row.appendChild(bubble);
  msgEl.appendChild(row);
  msgEl.scrollTop = msgEl.scrollHeight;
}

function addSuggestions() {
  const wrap = document.createElement("div");
  wrap.className = "suggestions";
  SUGGESTED.forEach(q => {
    const btn = document.createElement("button");
    btn.className = "suggestion-btn"; btn.textContent = q;
    btn.onclick = () => { wrap.remove(); send(q); };
    wrap.appendChild(btn);
  });
  msgEl.appendChild(wrap);
  msgEl.scrollTop = msgEl.scrollHeight;
}

async function send(text) {
  const msg = (text || inputEl.value).trim();
  if (!msg) return;
  inputEl.value = "";
  sendBtn.disabled = true;
  addMessage("user", msg);
  history.push({ role: "user", parts: [{ text: msg }] });
  addTyping();

  try {
    const res = await fetch(API_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        system_instruction: { parts: [{ text: SYSTEM_PROMPT }] },
        contents: history
      })
    });
    const data = await res.json();
    const reply = data.candidates?.[0]?.content?.parts?.[0]?.text || "응답을 받을 수 없습니다.";
    document.getElementById("typing")?.remove();
    addMessage("bot", reply);
    history.push({ role: "model", parts: [{ text: reply }] });
  } catch {
    document.getElementById("typing")?.remove();
    addMessage("bot", "오류가 발생했습니다. API 키를 확인하거나 잠시 후 다시 시도해 주세요.");
  }
  sendBtn.disabled = false;
  inputEl.focus();
}

sendBtn.onclick = () => send();
